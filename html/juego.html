// Piedra, papel o tijera
const ICONOS = { piedra: "✊", papel: "✋", tijera: "✌️" };

function obtenerEleccionAleatoria() {
    const opciones = ['piedra', 'papel', 'tijera'];
    return opciones[Math.floor(Math.random() * opciones.length)];
}

let partidas = 0;
let ganadas = 0;
let empatadas = 0;
let perdidas = 0;

// Repite una animación CSS aunque la clase ya estuviera puesta
function animar(elemento, clase) {
    elemento.classList.remove(clase);
    void elemento.offsetWidth; // fuerza el reinicio de la animación
    elemento.classList.add(clase);
}

function piedraPapelTijera(jugador) {
    const maquina = obtenerEleccionAleatoria();
    const pantalla = document.getElementById("pantalla");
    let estado;      // "gana", "pierde" o "empate": el CSS lo usa para el color y la animación
    let contador;    // marcador que sube en esta partida
    partidas++;

    if (jugador === maquina) {
        pantalla.value = "Empate";
        estado = "empate";
        contador = "empatadas";
        empatadas++;
    } else if (
        (jugador === 'piedra' && maquina === 'tijera') ||
        (jugador === 'tijera' && maquina === 'papel') ||
        (jugador === 'papel' && maquina === 'piedra')
    ) {
        pantalla.value = "Gana el jugador";
        estado = "gana";
        contador = "ganadas";
        ganadas++;
    } else {
        pantalla.value = "Gana la máquina";
        estado = "pierde";
        contador = "perdidas";
        perdidas++;
    }

    pantalla.dataset.estado = estado;
    animar(pantalla, "anima");

    document.getElementById("jugada-jugador").innerHTML = `Tu elección: <span class="icono">${ICONOS[jugador]}</span> ${jugador}`;
    document.getElementById("jugada-maquina").innerHTML = `Elección de la máquina: <span class="icono">${ICONOS[maquina]}</span> ${maquina}`;
    document.getElementById("partidas").textContent = `Partidas jugadas: ${partidas}`;
    document.getElementById("ganadas").textContent = `Partidas ganadas: ${ganadas}`;
    document.getElementById("empatadas").textContent = `Partidas empatadas: ${empatadas}`;
    document.getElementById("perdidas").textContent = `Partidas perdidas: ${perdidas}`;

    animar(document.getElementById(contador), "destello");
    animar(document.getElementById("partidas"), "destello");
}
